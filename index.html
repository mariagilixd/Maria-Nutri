Sincronización por Alias (Google Sheets)

Sin contraseñas: escribe tu alias y sincroniza entre iPhone y Mac

Al pegar esta URL una vez, cada comida o entreno se guardará automáticamente en tu hoja de Google Sheets.

📋 Ver código para pegar en tu Google Sheets (1 minuto)
  1. Crea una hoja de cálculo nueva en Google Sheets.
  2. En el menú superior ve a Extensiones > Apps Script.
  3. Borra lo que haya, pega el código de abajo y dale a Implementar > Nueva implementación.
  4. Elige tipo Aplicación web, ejecutar como "Yo" y quién tiene acceso: "Cualquier usuario". Copia la URL que te da y pégala arriba.
function doGet(e) {
  var alias = (e.parameter.alias || 'maria-nutri').toLowerCase();
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === alias) {
      return ContentService.createTextOutput(data[i][1])
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
  return ContentService.createTextOutput(JSON.stringify({empty: true}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  var alias = (body.alias || 'maria-nutri').toLowerCase();
  var payload = JSON.stringify(body.data);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Alias', 'JSON_Data', 'Ultima_Actualizacion']);
  }
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === alias) {
      sheet.getRange(i + 1, 2).setValue(payload);
      sheet.getRange(i + 1, 3).setValue(new Date());
      return ContentService.createTextOutput(JSON.stringify({ok: true}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
  sheet.appendRow([alias, payload, new Date()]);
  return ContentService.createTextOutput(JSON.stringify({ok: true}))
    .setMimeType(ContentService.MimeType.JSON);
}
